168 std::string Generator::single_sentence(const ArgVec2& wordArg,
169 const ArgVec2& frag)
170 {
171 static const std::vector<int> weights = {88, 7, 3, 2};
172 static const std::vector<int> weightsEndMark = {80, 15, 5};
173 static const std::vector<std::string> mapped = {", ",
174 "; ",
175 ": ",
176 " - "};
177 static const std::vector<std::string> mappedEndMark = {".", "?", "!"};
178 std::string result;
179 int words;
180 int frags = frag.roll(m_Gen);
181 for (int i = 0; i < frags; ++i)
182 {
183 words = wordArg.roll(m_Gen);
184 result += word(words);
185 int check = weighted_random_idx(weights);
186 // don't do if only one fragment
187 if (i != frags - 1)
188 {
189 result += mapped.at(check);
190 }
191 }
192 result += mappedEndMark.at(weighted_random_idx(weightsEndMark));
193 result.at(0) = LPSM_SAFE_CCTYPE(char, std::toupper, result.at(0));
194 return result;
195 }

197 std::string Generator::single_fmt_paragraph(bool useLipsum, bool useHtml)
198 {
199 std::string ret;
200 int sents = m_Settings.sent.roll(m_Gen);
201 int fmtRoll;
202 bool addLink;
203 bool isBold;
204 constexpr int FMT_PARA_CHANCE_FMT = 14;
205
206 // 1 in 15
207 static const std::vector<int> weights = {FMT_PARA_CHANCE_FMT, 1};
208
209 if (useHtml)
210 {
211 ret += "<p>";
212 }
213 for (int i = 0; i < sents; ++i)
214 {
215 fmtRoll = weighted_random_idx(weights);
216 if (fmtRoll == 1)
217 {
218 addLink = LPSM_FLIP_COIN;
219 isBold = LPSM_FLIP_COIN;
220 }
221 if (i == 0 && useLipsum)
222 {
223 ret += GenerateDefaultLipsumSentence();
224 }
225 else if ((fmtRoll == 1) && addLink)
226 {
227 ret += fmt_link(useHtml);
228 }
229 else if ((fmtRoll == 1) && !addLink)
230 {
231 ret += fmt_emphasis(isBold, useHtml);
232 }
233 else
234 {
235 ret += single_sentence(m_Settings.word, m_Settings.frag);
236 }
237 ret += " ";
238 }
239 if (useHtml)
240 {
241 ret += "</p>";
242 }
243 ret += "\n\n";
244 return ret;
245 }
