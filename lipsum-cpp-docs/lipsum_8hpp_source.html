696 const ArgVec2& level,

699{

702 ret += lpsm::GenerateMarkdownParagraph(word,

706 fragFmt,
707 wordHead,
708 linkURL,
709 false);
710 numElements -= 2;

712 bool ordered;

714 lpsm::ArgVec2 order = ArgVec2(0, 1);
715 while (numElements > 0)

718 ordered = static_cast<bool>(order.Roll());

724 frag,

730 false);

733 case 1:

797std::string lpsm::GenerateMarkdownParagraph(const lpsm::ArgVec2& word,
798 const lpsm::ArgVec2& frag,
799 const lpsm::ArgVec2& sent,
800 const ArgVec2& wordFmt,
801 const ArgVec2& fragFmt,
802 const ArgVec2& wordLink,
803 const std::string& linkURL,
804 bool useLipsum)
805{
806 std::string ret;
807 int sents = sent.Roll();
808 int fmtRoll;
809 bool addLink;
810 bool isBold;
811 lpsm::ArgVec2 linkEmph = ArgVec2(0, 1);
812 lpsm::ArgVec2 boldItal = ArgVec2(0, 1);
813 for (int i = 0; i < sents; ++i)
814 {
815 fmtRoll = sent.Roll();
816 addLink = static_cast<bool>(linkEmph.Roll());
817 isBold = static_cast<bool>(boldItal.Roll());
818 if (i == 0 && useLipsum)
819 {
820 ret += lpsm::GenerateDefaultLipsumSentence();
821 }
822 else if ((fmtRoll == sent.min) && addLink)
823 {
824 ret += lpsm::GenerateMarkdownLink(linkURL,
825 wordFmt,
826 fragFmt,
827 wordLink);
828 }
829 else if ((fmtRoll == sent.min) && !addLink)
830 {
831 ret += lpsm::GenerateMarkdownEmphasis(isBold, wordFmt, fragFmt);
832 }
833 else
834 {
835 ret += lpsm::GenerateSentence(word, frag);
836 }
837 ret += " ";
838 }
839 ret += "\n\n";
840 return ret;
841}
