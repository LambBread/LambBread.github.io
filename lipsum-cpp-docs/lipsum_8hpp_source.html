312std::string lipsum::internal::HandleHTMLEntity(const std::string& str)
313{
314 std::string result;
315 for(const auto& c : str)
316 {
317 switch(c)
318 {
319 case '&':
320 {
321 result += "&amp;";
322 break;
323 }
324 case '<':
325 {
326 result += "&lt;";
327 break;
328 }
329 case '>':
330 {
331 result += "&gt;";
332 break;
333 }
334 case '"':
335 {
336 result += "&quot;";
337 break;
338 }
339 case '\'':
340 {
341 result += "&apos;";
342 break;
343 }
344 default:
345 {
346 result.push_back(c);
347 }
348 }
349 }
350 return result;
351}

430std::string lipsum::GenerateSentence(const lipsum::ArgVec2& word, const lipsum::ArgVec2& frag)
431{
432 constexpr int CHANCE_COMMA = 97;
433 constexpr int CHANCE_SEMI = 9;
434 std::string result;
435 int words = word.Roll();
436 int frags = frag.Roll();
437 static std::random_device rd;
438 static std::mt19937 gen(rd());
439 std::uniform_int_distribution<> dist(0, 100);
440 for(int i = 0; i < frags; ++i)
441 {
442 result += lipsum::GenerateSentenceFragment();
443 int check = dist(gen);
444 if(i != frags - 1)
445 {
446 if(dist(gen) < CHANCE_SEMI)
447 {
448 result += "; ";
449 }
450 else if(dist(gen) < CHANCE_COMMA)
451 {
452 result += ", ";
453 }
454 else
455 {
456 result += " - ";
457 }
458 }
459 }
460 result += ".";
461 result.at(0) = std::toupper(result.at(0));
462 return result;
