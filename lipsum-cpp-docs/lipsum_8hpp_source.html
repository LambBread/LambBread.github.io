500std::string lpsm::internal::HandleHTMLEntity(const std::string& str)
501{
502 std::string result;
503 for(const auto& c : str)
504 {
505 switch(c)
506 {
507 case '&':
508 {
509 result += "&amp;";
510 break;
511 }
512 case '<':
513 {
514 result += "&lt;";
515 break;
516 }
517 case '>':
518 {
519 result += "&gt;";
520 break;
521 }
522 case '"':
523 {
524 result += "&quot;";
525 break;
526 }
527 case '\'':
528 {
529 result += "&apos;";
530 break;
531 }
532 default:
533 {
534 result.push_back(c);
535 }
536 }
537 }
538 return result;

623std::string lpsm::GenerateSentence(const lpsm::ArgVec2& word, const lpsm::ArgVec2& frag)
624{
625 constexpr int CHANCE_COMMA = 97;
626 constexpr int CHANCE_SEMI = 9;
627 std::string result;
628 int words = word.Roll();
629 int frags = frag.Roll();
630 //static std::random_device rd;
631 //static std::mt19937 gen(rd());
632 //std::uniform_int_distribution<> dist(0, 100);
633 for(int i = 0; i < frags; ++i)
634 {
635 result += lpsm::GenerateSentenceFragment(word);
636 int check = lpsm::internal::RandomNumber(0, 100);
637 // don't do if only one fragment
638 if(i != frags - 1)
639 {
640 if(check < CHANCE_SEMI)
641 {
642 result += "; ";
643 }
644 else if(check < CHANCE_COMMA)
645 {
646 result += ", ";
647 }
648 else
649 {
650 result += " - ";
651 }
652 }
653 }
654 result += ".";
655 result.at(0) = std::toupper(result.at(0));
656 return result;
657}

662 const lpsm::ArgVec2& sent,
663 bool useLipsum)
664{
665 std::string result = "\t";

667 if(!useLipsum)
668 {
669 for(int i = 0; i < sents; ++i)
670 {
671 result += lpsm::GenerateSentence(word, frag) += " ";
672 }
673 }
674 else
675 {
676 result += lipsum::GenerateDefaultLipsumSentence() += " ";
677 for(int i = 0; i < sents - 1; ++i)
678 {
679 result += lpsm::GenerateSentence(word, frag) += " ";
680 }
681 }
682 // remove trailing space
683 result.pop_back();
684 result += "\n";
685 return result;
