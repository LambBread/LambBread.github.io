141std::string lipsum::GenerateSentence(int minWord, int maxWord)
142{
143 std::string result;
144 static std::random_device rd;
145 static std::mt19937 gen(rd());
146 std::uniform_int_distribution<> dist2(minWord, maxWord);
147 int words = dist2(gen);
148 bool includecomma = (words >= (maxWord * 4) / 5);
149 for(int i = 0; i < words; ++i)
150 {
151 result += lipsum::GenerateWord();
152 if(includecomma && i == (words / 2) - 1)
153 {
154 result += ",";
155 }
156 if(i != words - 1)
157 {
158 result += " ";
159 }
160 }
161 result += ".";
162 result.at(0) = std::toupper(result.at(0));
163 return result;
