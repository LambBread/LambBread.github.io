433int lipsum::internal::RandomNumber(int min, int max, int seed)
434{
435 static std::random_device rd;
436 if(seed == 0)
437 {
438 std::mt19937 gen(rd());
439 std::uniform_int_distribution<> dist(min, max);
440 return dist(gen);
441 }
442 else
443 {
444 std::mt19937 gen(seed);
445 std::uniform_int_distribution<> dist(min, max);
446 return dist(gen);
447 }
448}

513std::string lipsum::internal::HandleHTMLEntity(const std::string& str)
514{
515 std::string result;
516 for(const auto& c : str)
517 {
518 switch(c)
519 {
520 case '&':
521 {
522 result += "&amp;";
523 break;
524 }
525 case '<':
526 {
527 result += "&lt;";
528 break;
529 }
530 case '>':
531 {
532 result += "&gt;";
533 break;
534 }
535 case '"':
536 {
537 result += "&quot;";
538 break;
539 }
540 case '\'':
541 {
542 result += "&apos;";
543 break;
544 }
545 default:
546 {
547 result.push_back(c);
548 }
549 }
550 }
551 return result;

635std::string lipsum::GenerateSentence(const lipsum::ArgVec2& word, const lipsum::ArgVec2& frag, int seed)
636{
637 constexpr int CHANCE_COMMA = 97;
638 constexpr int CHANCE_SEMI = 9;
639 std::string result;
640 int words = word.Roll(seed);
641 int frags = frag.Roll(seed);
642 //static std::random_device rd;
643 //static std::mt19937 gen(rd());
644 //std::uniform_int_distribution<> dist(0, 100);
645 for(int i = 0; i < frags; ++i)
646 {
647 result += lipsum::GenerateSentenceFragment(word, seed);
648 int check = lipsum::internal::RandomNumber(0, 100, seed);
649 // don't do if only one fragment
650 if(i != frags - 1)
651 {
652 if(check < CHANCE_SEMI)
653 {
654 result += "; ";
655 }
656 else if(check < CHANCE_COMMA)
657 {
658 result += ", ";
659 }
660 else
661 {
662 result += " - ";
663 }
664 }
665 }
666 result += ".";
667 result.at(0) = std::toupper(result.at(0));
668 return result;
669}
