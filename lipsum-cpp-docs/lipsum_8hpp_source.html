123 std::string result;

125 {
126 switch(c)

207std::string lipsum::GenerateSentence(int minWord, int maxWord)
208{
209 std::string result;
210 static std::random_device rd;
211 static std::mt19937 gen(rd());
212 std::uniform_int_distribution<> dist2(minWord, maxWord);
213 int words = dist2(gen);
214 bool includecomma = (words >= (maxWord * 4) / 5);
215 for(int i = 0; i < words; ++i)
216 {
217 result += lipsum::GenerateWord();
218 if(includecomma && i == (words / 2) - 1)
219 {
220 result += ",";
221 }
222 if(i != words - 1)
223 {
224 result += " ";
225 }
226 }
227 result += ".";
228 result.at(0) = std::toupper(result.at(0));
229 return result;
230}
