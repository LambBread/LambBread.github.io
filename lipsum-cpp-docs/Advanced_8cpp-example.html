// A small helper to replace all occurrences of a token in a string.
static std::string
ReplaceAll(std::string s, const std::string& from, const std::string& to)
{
if (from.empty())
return s;
size_t pos = 0;
while ((pos = s.find(from, pos)) != std::string::npos)
{
s.replace(pos, from.length(), to);
pos += to.length();
}
return s;
}
