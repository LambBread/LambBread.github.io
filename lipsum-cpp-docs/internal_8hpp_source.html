189 template <typename T> T ToType(const std::string& input)
190 {
191 T res{};
192
193 if constexpr (std::same_as<T, std::string>)
194 {
195 return input;
196 }
197 else if constexpr (std::same_as<T, bool>)
198 {
199 // std::cout << "bool ToType\n";
200 return (input == "true" || input == "1");
201 }
202 else if constexpr (std::same_as<T, char>)
203 {
204 // std::cout << "char ToType\n";
205 return input.empty() ? '\0' : input.at(0);
206 }
207 else if constexpr (IsInt<T>)
208 {
209 // std::cout << "integer ToType\n";
210 auto [ptr, ec] = std::from_chars(input.data(),
211 input.data() + input.size(),
212 res);
213 if (ec != std::errc{})
214 {
215 LogWarn("lpsm::internal::ToType: std::from_chars() failed; "
216 "invalidly formatted integer? Returning default T.");
217 return T{};
218 }
219 return res;
220 }
221 else
222 // assume streamable
223 {
224 // std::cout << "generic ToType\n";
225 std::istringstream iss(input);
226 iss >> res;
227 return res;
228 }
229 }
