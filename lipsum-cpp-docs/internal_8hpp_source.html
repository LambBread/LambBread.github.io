268 template <typename T> T ToType(const std::string& input)
269 {
270 T res{};
271
272 if constexpr (std::same_as<T, std::string>)
273 {
274 return input;
275 }
276 else if constexpr (std::same_as<T, bool>)
277 {
278 // std::cout << "bool ToType\n";
279 return (input == "true" || input == "1");
280 }
281 else if constexpr (std::same_as<T, char>)
282 {
283 // std::cout << "char ToType\n";
284 return input.empty() ? '\0' : input.at(0);
285 }
286 else if constexpr (IsInt<T>)
287 {
288 // std::cout << "integer ToType\n";
289 auto [ptr, ec] = std::from_chars(input.data(),
290 input.data() + input.size(),
291 res);
292 if (ec != std::errc{})
293 {
294 LogWarn(internal::LogType::Error,
295 "lpsm::internal::ToType: std::from_chars() failed; "
296 "invalidly formatted integer? Returning default T.");
297 return T{};
298 }
299 return res;
300 }
301 else
302 // assume streamable
303 {
304 // std::cout << "generic ToType\n";
305 std::istringstream iss(input);
306 iss >> res;
307 return res;
308 }
