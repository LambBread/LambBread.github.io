173 template <UniformDistributionType T> T RandomNumber(T min, T max)
174 {
175 static thread_local std::mt19937 gen(std::random_device{}());
176 if (min > max)
177 {
178 T tempMax = max;
179 max = min;
180 min = tempMax;
181 }
182 if constexpr (std::is_integral_v<T>)
183 {
184 std::uniform_int_distribution<T> dist(min, max);
185 return dist(gen);
186 }
187 else
188 {
189 std::uniform_real_distribution<T> dist(min, max);
190 return dist(gen);
191 }
192 }

283 template <typename T> T ToType(const std::string& input)
284 {
285 T res{};
286
287 if constexpr (std::same_as<T, std::string>)
288 {
289 return input;
290 }
291 else if constexpr (std::same_as<T, bool>)
292 {
293 // std::cout << "bool ToType\n";
294 return (input == "true" || input == "1");
295 }
296 else if constexpr (std::same_as<T, char>)
297 {
298 // std::cout << "char ToType\n";
299 return input.empty() ? '\0' : input.at(0);
300 }
301 else if constexpr (IsInt<T>)
302 {
303 // std::cout << "integer ToType\n";
304 auto [ptr, ec] = std::from_chars(input.data(),
305 input.data() + input.size(),
306 res);
307 if (ec != std::errc{})
308 {
309 LogWarn("lpsm::internal::ToType: std::from_chars() failed; "
310 "invalidly formatted integer? Returning default T.");
311 return T{};
312 }
313 return res;
314 }
315 else
316 // assume streamable
317 {
318 // std::cout << "generic ToType\n";
319 std::istringstream iss(input);
320 iss >> res;
321 return res;
322 }
323 }
