184 template <typename T> T ToType(const std::string& input)
185 {
186 T res{};
187
188 if constexpr (std::same_as<T, std::string>)
189 {
190 return input;
191 }
192 else if constexpr (std::same_as<T, bool>)
193 {
194 // std::cout << "bool ToType\n";
195 return (input == "true" || input == "1");
196 }
197 else if constexpr (std::same_as<T, char>)
198 {
199 // std::cout << "char ToType\n";
200 return input.empty() ? '\0' : input.at(0);
201 }
202 else if constexpr (IsInt<T>)
203 {
204 // std::cout << "integer ToType\n";
205 auto [ptr, ec] = std::from_chars(input.data(),
206 input.data() + input.size(),
207 res);
208 if (ec != std::errc{})
209 {
210 LogWarn("lpsm::internal::ToType: std::from_chars() failed; "
211 "invalidly formatted integer? Returning default T.");
212 return T{};
213 }
214 return res;
215 }
216 else
217 // assume streamable
218 {
219 // std::cout << "generic ToType\n";
220 std::istringstream iss(input);
221 iss >> res;
222 return res;
223 }
224 }
